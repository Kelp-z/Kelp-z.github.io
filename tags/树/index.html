B树

B树 每个节点可以有多个子节点。假设m阶B树（m = 2t，t为最小度数），则： 每个节点最多有m个孩子（2t个） 除根节点外，每个节点至少有t个孩子 每个节点的关键字数量在[t-1, 2t-1]之间 所有叶子节点在同一层 节点内的关键字按升序排列 class BTree{ class BTreeNode{ private int[] keys; private BTreeNode[] children; private int t;//最小度数 private int n;//关键字的数量 private boolean leaf; BTreeNode(int t,boolean leaf){ this.t = t; this.leaf = leaf; this.keys = new int[2*t - 1]; this.children = new BTreeNode[2*t]; this.n = 0; } } int t; Node root; public BTree(int t){ this.t = t; this.root = new Node(t,true);//根节点初始为叶节点 } public BTreeNode search(int k) { return search(root, k); } private BTreeNode search(BTreeNode node, int k) { // 1. 找到第一个大于等于k的关键字的索引 int i = 0; while (i < node.n && k > node.keys[i]) { i++; } // 2. 如果找到匹配的关键字，返回当前节点 if (i < node.n && k == node.keys[i]) { return node; } // 3. 如果当前是叶子节点且没找到，说明k不存在 if (node.leaf) { return null; } // 4. 递归查找对应的子节点 return search(node.children[i], k); } public void insert(int k) { BTreeNode r = root; // 情况1：根节点已满（触发分裂） if (r.n == 2 * t - 1) { // 创建新的根节点（B树高度+1） BTreeNode newRoot = new BTreeNode(t, false); // 原根节点变为新根的第一个孩子 newRoot.children[0] = r; // 分裂原根节点 splitChild(newRoot, 0, r); // 确定新根节点的插入位置（左或右子树） int i = 0; if (newRoot.keys[0] < k) { i++; } // 插入到对应的非满子节点中 insertNonFull(newRoot.children[i], k); // 更新根节点 root = newRoot; } // 情况2：根节点未满，直接插入 else { insertNonFull(r, k); } } /** * 分裂满的子节点（B树的核心操作！） * @param parent 父节点 * @param childIndex 要分裂的子节点在父节点children数组中的索引 * @param fullChild 已满的子节点（关键字数=2t-1） */ private void splitChild(BTreeNode parent, int childIndex, BTreeNode fullChild) { // 1. 创建新节点存储分裂后的右半部分 BTreeNode newNode = new BTreeNode(t, fullChild.leaf); // 分裂后每个节点的关键字数 = t-1 int mid = t - 1; // 2. 将fullChild的后t-1个关键字复制到newNode for (int i = 0; i < mid; i++) { newNode.keys[i] = fullChild.keys[i + t]; newNode.n++; // newNode的关键字数增加 } // 3. 如果fullChild不是叶子节点，复制后t个孩子到newNode if (!fullChild.leaf) { for (int i = 0; i < t; i++) { newNode.children[i] = fullChild.children[i + t]; } } // 4. 将fullChild的关键字数更新为t-1（只保留前t-1个） fullChild.n = mid; // 5. 父节点的children数组腾出位置，插入newNode for (int i = parent.n; i > childIndex; i--) { parent.children[i + 1] = parent.children[i]; } parent.children[childIndex + 1] = newNode; // 6. 父节点的keys数组腾出位置，插入fullChild的中间关键字 for (int i = parent.n - 1; i >= childIndex; i--) { parent.keys[i + 1] = parent.keys[i]; } parent.keys[childIndex] = fullChild.keys[mid]; // 中间关键字提升到父节点 parent.n++; // 父节点的关键字数增加 } /** * 向非满节点插入关键字 * @param node 非满节点 * @param k 要插入的关键字 */ private void insertNonFull(BTreeNode node, int k) { // 从后往前找插入位置（保持关键字有序） int i = node.n - 1; // 情况1：当前是叶子节点，直接插入 if (node.leaf) { // 后移比k大的关键字，腾出位置 while (i >= 0 && node.keys[i] > k) { node.keys[i + 1] = node.keys[i]; i--; } // 插入关键字 node.keys[i + 1] = k; node.n++; } // 情况2：当前是非叶子节点，递归找子节点插入 else { // 找到对应的子节点索引 while (i >= 0 && node.keys[i] > k) { i--; } i++; // 检查子节点是否已满，如果满则先分裂 if (node.children[i].n == 2 * t - 1) { splitChild(node, i, node.children[i]); // 分裂后，可能需要调整插入位置（因为父节点新增了关键字） if (node.keys[i] < k) { i++; } } // 插入到子节点 insertNonFull(node.children[i], k); } } }
