@CallerSensitive
    public static Object newProxyInstance(ClassLoader loader,
                                          Class<?>[] interfaces,
                                          InvocationHandler h) {
 //先调用了requireNonNull(h)方法，判断InvocationHandler对象是否为空，若为空则抛出空指针异常                                  InvocationHandler h)
        Objects.requireNonNull(h);

/*这一段代码是在检查是否有安全管理器，如果没有则把caller设置为空，如果有就调用 Reflection.getCallerClass()获取当前调用newProxyInstance方法的那个类。
  在java早期，安全管理器会控制代码权限，它会检查哪个类在调用newProxyInstance，这个类有没有权限在某个包下创建代理类。现在安全管理器基本上没人用了。所以这行段代码在大部分情况下，caller都等于null
  */                                          
        @SuppressWarnings("removal")
        final Class<?> caller = System.getSecurityManager() == null
                                    ? null
                                    : Reflection.getCallerClass();


                                          
  //获取代理类的构造器                                  
        Constructor<?> cons = getProxyConstructor(caller, loader, interfaces);

  //创建代理对象并返回                                        
        return newProxyInstance(caller, cons, h);
    }


//对getProxyConstructor方法的分析
 private static Constructor<?> getProxyConstructor(Class<?> caller,
                                                      ClassLoader loader,
                                                      Class<?>... interfaces)
    {
        //只有1个接口的情况
        if (interfaces.length == 1) {
            Class<?> intf = interfaces[0];
            //权限检验，caller大部分情况为空，checkProxyAccess通常不会执行，在上面解释过
            if (caller != null) {
                checkProxyAccess(caller, loader, intf);
            }
            
            /*
            首先，我们要先知道return这一段在干什么——以类加载器和接口为key，去缓存里查找代理类的构造器，如果缓存里没有，就生成对应的构造器存入缓存，并返回构造器
            接下来我们再具体分析
            proxyCache是一个双层缓存，第一层key是你要代理的接口（intf），第二层是类加载器（loader）
            computeIfAbsent方法是Map接口中的一个方法，它的作用是，如果Map中有这个key，就返回对应的Value；如果没有，就生成一个Value，存入Map，并返回
            lambda表达式中的参数ld是类加载器loader，clv.key（）可以获取到传入的接口intf
            那么这段代码我们可以解释为，先通过 proxyCache.sub(intf)找到属于这个接口的子缓存，在调用computeIfAbsent方法以类加载器（loader）作为key去查对应的ProxyBuilder对象，如果缓存里没有，就调用这个lambda表达式(ld, clv) -> new ProxyBuilder(ld, clv.key()）创建一个ProxyBuilder对象，最后在通过.build()方法获取到对应的构造器。
            */
            return proxyCache.sub(intf).computeIfAbsent(
                loader,
                (ld, clv) -> new ProxyBuilder(ld, clv.key()).build()
            );
        } else {
            //有多个接口的情况，与上面只有1个接口的情况相似
            final Class<?>[] intfsArray = interfaces.clone();
            if (caller != null) {
                checkProxyAccess(caller, loader, intfsArray);
            }
            final List<Class<?>> intfs = Arrays.asList(intfsArray);
            return proxyCache.sub(intfs).computeIfAbsent(
                loader,
                (ld, clv) -> new ProxyBuilder(ld, clv.key()).build()
            );
        }
    }



//创建代理对象的方法
 private static Object newProxyInstance(Class<?> caller, // null if no SecurityManager
                                           Constructor<?> cons,
                                           InvocationHandler h) {
        /*
         * Invoke its constructor with the designated invocation handler.
         */
        try {
            if (caller != null) {
                checkNewProxyPermission(caller, cons.getDeclaringClass());
            }
            //这一行代码是通过反射获取了代理类的唯一构造方法，把你写的InvocationHandler实例也就是h作为参数传入，这个h会  				被保存到父类Proxy的成员变量h中，最后创建并返回这个代理类的实例对象
            return cons.newInstance(new Object[]{h});//整段代码的核心部分，其它都是异常处理。
        } catch (IllegalAccessException | InstantiationException e) {
            throw new InternalError(e.toString(), e);
        } catch (InvocationTargetException e) {
            Throwable t = e.getCause();
            if (t instanceof RuntimeException re) {
                throw re;
            } else {
                throw new InternalError(t.toString(), t);
            }
        }
    }
